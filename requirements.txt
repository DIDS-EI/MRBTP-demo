# Minimal runtime dependencies for the mrbtp-demo standalone package.
#
# This list was derived empirically: every third-party top-level package that
# the trimmed `mrbtp` subset + `mrbtp_demo` actually import.  The heavyweight
# extras from the original mrbtp repo (openai, hydra-core, sympy, matplotlib,
# opencv, networkx, tabulate, latextable, ipdb, tqdm, requests, tdw, ...) are
# NOT needed for this demo and have been removed.

# --- core simulation / planning ---
gymnasium~=0.29.1          # RL environment API
minigrid~=2.3.1            # the 2-D grid world the demo runs on
py_trees~=2.2.3            # behavior-tree runtime
numpy>=1.19.3              # grid math / observations
antlr4-python3-runtime     # parses the BTML behavior-tree DSL

# --- rendering / drawing ---
pygame>=2.1                # MiniGrid's window backend (headless via SDL dummy)
pydot~=1.4                 # renders behavior trees to graphs (BehaviorTree.draw)
Pillow>=9.0                # tile icons for the grid renderer

# --- misc helpers used by mrbtp.utils ---
shortuuid~=1.0.11          # stable short ids for tree nodes
pandas~=2.2.2              # a couple of tabular helpers in mrbtp.utils.tools
